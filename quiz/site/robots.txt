# quiz.dobsinsky.dev - no search indexing, but link previews must still work.
#
# Why not indexed: a search result pointing at a live health survey invites
# respondents who were never approached personally and arrive with no ?src=
# attribution, which breaks the sampling the thesis relies on. index.html also
# carries <meta name="robots" content="noindex, nofollow"> for crawlers that
# fetch the page but never read this file.
#
# Why the preview bots are allowed anyway: they honour robots.txt, and a blanket
# Disallow would stop Messenger, WhatsApp and Instagram from reading the
# Open Graph tags - so a link pasted into the group chats this survey is actually
# distributed through would show a bare URL instead of a title and an image.
# These agents generate previews; they do not index.
#
# Most specific User-agent group wins, so the allows below beat the catch-all.

User-agent: facebookexternalhit
Allow: /

Allow: /

Allow: /

User-agent: WhatsApp
Allow: /

Allow: /

Allow: /

Allow: /

Allow: /

Allow: /
